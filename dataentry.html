// --- FUNGSI LOAD DATA (GUNA FETCH UNTUK GITHUB) --- function loadUnitData() { var unit = document.getElementById('unit').value.trim(); if (!unit) { alert("Please enter a UNIT first."); return; } document.getElementById('message').innerText = "Loading data..."; document.getElementById('message').className = "mt-3 text-center fw-bold text-secondary"; // Tambah { redirect: "follow" } supaya Google Apps Script tidak disekat fetch(SCRIPT_URL + "?action=getUnitData&unit=" + encodeURIComponent(unit), { method: "GET", redirect: "follow" }) .then(response => { if (!response.ok) throw new Error("Network response was not ok"); return response.json(); }) .then(data => { if (data) { document.getElementById('statusDrip').value = data.statusDrip || "Inden Belum Diterima"; document.getElementById('statusBukanUbat').value = data.statusBukanUbat || "Inden Belum Diterima"; document.getElementById('jumlahTroli').value = data.jumlahTroli || ""; document.getElementById('perluTroli').value = data.perluTroli || "Tidak"; document.getElementById('emailPic').value = data.emailPic || ""; document.getElementById('telPic').value = data.telPic || ""; document.getElementById('message').innerText = "Data loaded for " + unit; document.getElementById('message').className = "mt-3 text-center fw-bold text-success"; } else { document.getElementById('message').innerText = "Unit not found. Fill the form to add as a new record."; document.getElementById('message').className = "mt-3 text-center fw-bold text-warning"; document.getElementById('statusDrip').value = "Inden Belum Diterima"; document.getElementById('statusBukanUbat').value = "Inden Belum Diterima"; document.getElementById('jumlahTroli').value = ""; document.getElementById('perluTroli').value = "Tidak"; document.getElementById('emailPic').value = ""; document.getElementById('telPic').value = ""; } updateSelectColor(document.getElementById('statusDrip')); updateSelectColor(document.getElementById('statusBukanUbat')); }) .catch(error => { // Paparkan ralat sebenar di skrin untuk memudahkan troubleshoot console.error("Fetch Error:", error); document.getElementById('message').innerText = "Error: " + error.message; document.getElementById('message').className = "mt-3 text-center fw-bold text-danger"; }); }
